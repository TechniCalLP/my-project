1️⃣ /middleware.ts - ตรวจสอบ Role ที่ Server Level
✅ ต้องทำ:
typescriptimport { withAuth } from "next-auth/middleware"
import { NextRequest, NextResponse } from "next/server"

export const middleware = withAuth(
  function middleware(req: NextRequest & { nextauth: any }) {
    const token = req.nextauth.token

    // ✅ ตรวจสอบ Admin routes
    if (req.nextUrl.pathname.startsWith("/admin")) {
      // ❌ ถ้า role ไม่ใช่ admin → redirect ออก
      if (!token || token.role !== "admin") {
        return NextResponse.redirect(new URL("/", req.url))
      }
    }

    // ✅ ตรวจสอบ Student routes
    if (req.nextUrl.pathname.startsWith("/dashboard")) {
      // ❌ ถ้า role ไม่ใช่ student → redirect ออก
      if (!token || token.role !== "student") {
        return NextResponse.redirect(new URL("/auth/login", req.url))
      }
    }

    return NextResponse.next()
  },
  {
    callbacks: {
      authorized: ({ token }) => {
        // ✅ ต้อง login ก่อน
        return !!token
      },
    },
  }
)

export const config = {
  matcher: [
    "/admin/:path*",
    "/dashboard/:path*",
    "/api/admin/:path*",
  ],
}

2️⃣ /app/(admin)/admin/layout.tsx - Client-side Guard
✅ ต้องทำ:
typescript'use client'

import { useSession } from "next-auth/react"
import { useRouter } from "next/navigation"
import { useEffect, useState } from "react"

export default function AdminLayout({
  children,
}: {
  children: React.ReactNode
}) {
  const { data: session, status } = useSession()
  const router = useRouter()
  const [isAuthorized, setIsAuthorized] = useState(false)

  useEffect(() => {
    // ❌ ถ้า loading ยัง → ไม่ render อะไร
    if (status === "loading") {
      return
    }

    // ❌ ถ้า session ไม่มี → redirect
    if (!session) {
      router.push("/auth/login")
      return
    }

    // ❌ ถ้า role ไม่ใช่ admin → redirect
    if (session.user?.role !== "admin") {
      console.warn("Unauthorized access attempt to admin panel")
      router.push("/")
      return
    }

    // ✅ ตรวจสอบผ่าน → render
    setIsAuthorized(true)
  }, [session, status, router])

  // ❌ ถ้า unauthorized → ไม่ render
  if (!isAuthorized) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="text-center">
          <p className="text-lg text-red-600">กำลังตรวจสอบสิทธิ์...</p>
        </div>
      </div>
    )
  }

  return <>{children}</>
}

// ✅ Protect ทุก Admin API
export async function GET(req: NextRequest) {
  const session = await getServerSession(authOptions)

  // ❌ ถ้า session ไม่มี
