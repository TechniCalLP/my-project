```typescript
"use client"

import { useState } from "react"
import { useRouter } from "next/navigation"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { useToast } from "@/components/ui/use-toast"
import { Upload, Loader2, CheckCircle2, XCircle, Download } from "lucide-react"
import { Card, CardContent } from "@/components/ui/card"

interface StudentRow {
  studentId: string
  firstName: string
  lastName: string
  department: string
  year: number
  email?: string
  phone?: string
  status?: 'pending' | 'success' | 'error'
  error?: string
}

export default function ImportStudentsForm() {
  const router = useRouter()
  const { toast } = useToast()
  const [file, setFile] = useState(null)
  const [loading, setLoading] = useState(false)
  const [preview, setPreview] = useState([])
  const [importing, setImporting] = useState(false)

  const handleFileChange = async (e: React.ChangeEvent) => {
    const selectedFile = e.target.files?.[0]
    if (!selectedFile) return

    // Validate file type
    const validTypes = ['text/csv', 'application/vnd.ms-excel', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet']
    if (!validTypes.includes(selectedFile.type) && !selectedFile.name.endsWith('.csv')) {
      toast({
        title: "ไฟล์ไม่ถูกต้อง",
        description: "กรุณาเลือกไฟล์ CSV หรือ Excel",
        variant: "destructive",
      })
      return
    }

    setFile(selectedFile)
    setLoading(true)

    try {
      // Read file
      const text = await selectedFile.text()
      const rows = parseCSV(text)
      
      if (rows.length === 0) {
        throw new Error('ไฟล์ว่างเปล่า')
      }

      setPreview(rows)
      toast({
        title: "อ่านไฟล์สำเร็จ",
        description: `พบข้อมูล ${rows.length} รายการ`,
      })
    } catch (error: any) {
      toast({
        title: "เกิดข้อผิดพลาด",
        description: error.message || "ไม่สามารถอ่านไฟล์ได้",
        variant: "destructive",
      })
      setFile(null)
      setPreview([])
    } finally {
      setLoading(false)
    }
  }

  const parseCSV = (text: string): StudentRow[] => {
    const lines = text.trim().split('\n')
    if (lines.length < 2) {
      throw new Error('ไฟล์ต้องมีอย่างน้อย 2 บรรทัด (Header + Data)')
    }

    // Skip header
    const dataLines = lines.slice(1)
    
    return dataLines.map((line, index) => {
      const values = line.split(',').map(v => v.trim().replace(/^"|"$/g, ''))
      
      if (values.length < 5) {
        throw new Error(`บรรทัดที่ ${index + 2}: ข้อมูลไม่ครบ (ต้องมีอย่างน้อย 5 คอลัมน์)`)
      }

      return {
        studentId: values[0],
        firstName: values[1],
        lastName: values[2],
        department: values[3],
        year: parseInt(values[4]),
        email: values[5] || undefined,
        phone: values[6] || undefined,
        status: 'pending' as const,
      }
    }).filter(row => row.studentId) // Filter out empty rows
  }

  const handleImport = async () => {
    if (preview.length === 0) return

    if (!confirm(`ต้องการ Import นักศึกษา ${preview.length} คน ใช่หรือไม่?`)) {
      return
    }

    setImporting(true)

    try {
      const response = await fetch('/api/admin/students/import', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ students: preview }),
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'Import failed')
      }

      toast({
        title: "Import สำเร็จ",
        description: `Import นักศึกษา ${result.success} คน สำเร็จ${result.failed > 0 ? `, ล้มเหลว ${result.failed} คน` : ''}`,
      })

      // Update preview with results
      setPreview(result.results)

      // Redirect after 2 seconds if all success
      if (result.failed === 0) {
        setTimeout(() => {
          router.push('/admin/students')
          router.refresh()
        }, 2000)
      }
    } catch (error: any) {
      toast({
        title: "เกิดข้อผิดพลาด",
        description: error.message || "ไม่สามารถ Import ได้",
        variant: "destructive",
      })
    } finally {
      setImporting(false)
    }
  }

  const downloadTemplate = () => {
    const template = `studentId,firstName,lastName,department,year,email,phone
66011234,สมชาย,ใจดี,คอมพิวเตอร์,1,somchai@email.com,0812345678
66011235,สมหญิง,รักเรียน,ช่างยนต์,2,somying@email.com,0823456789
66011236,สมศักดิ์,มานะ,ช่างไฟฟ้า,1,,0834567890`

    const blob = new Blob([template], { type: 'text/csv;charset=utf-8;' })
    const link = document.createElement('a')
    link.href = URL.createObjectURL(blob)
    link.download = 'template_students.csv'
    link.click()
  }

  return (
    
      {/* Upload Section */}
      
        
          
            
              
                เลือกไฟล์ CSV/Excel
              
              
                
                
                  
                  ดาวน์โหลดตัวอย่าง
                
              
              {file && (
                
                  ไฟล์: {file.name} ({(file.size / 1024).toFixed(2)} KB)
                
              )}
            
          
        
      

      {/* Preview Section */}
      {preview.length > 0 && (
        
          
            
              
                ตัวอย่างข้อมูล ({preview.length} รายการ)
              
              
                {importing ? (
                  <>
                    
                    กำลัง Import...
                  </>
                ) : (
                  <>
                    
                    Import ทั้งหมด
                  </>
                )}
              
            

            
              
                
                  
                    
                      #
                    
                    
                      รหัส
                    
                    
                      ชื่อ
                    
                    
                      นามสกุล
                    
                    
                      แผนก
                    
                    
                      ชั้นปี
                    
                    
                      สถานะ
                    
                  
                
                
                  {preview.map((row, index) => (
                    <tr key={index} className={row.status === 'error' ? 'bg-red-50' : ''}>
                      {index + 1}
                      {row.studentId}
                      {row.firstName}
                      {row.lastName}
                      {row.department}
                      
                        {row.year <= 3 ? `ปวช.${row.year}` : `ปวส.${row.year - 3}`}
                      
                      
                        {row.status === 'pending' && (
                          รอ Import
                        )}
                        {row.status === 'success' && (
                          
                            
                            สำเร็จ
                          
                        )}
                        {row.status === 'error' && (
                          
                            
                            {row.error}
                          
                        )}
                      
                    
                  ))}
                
              
            
          
        
      )}
    
  )
}
```

## 3. Create Import API

File: `app/api/admin/students/import/route.ts`

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/app/api/auth/[...nextauth]/route'
import { prisma } from '@/lib/prisma'
import bcrypt from 'bcryptjs'

interface StudentImport {
  studentId: string
  firstName: string
  lastName: string
  department: string
  year: number
  email?: string
  phone?: string
}

export async function POST(req: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    if (!session || session.user.role !== 'admin') {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const { students } = await req.json()

    if (!Array.isArray(students) || students.length === 0) {
      return NextResponse.json({ error: 'Invalid data' }, { status: 400 })
    }

    const results = []
    let successCount = 0
    let failedCount = 0

    for (const student of students) {
      try {
        // Validate required fields
        if (!student.studentId || !student.firstName || !student.lastName || !student.department || !student.year) {
          results.push({
            ...student,
            status: 'error',
            error: 'ข้อมูลไม่ครบ',
          })
          failedCount++
          continue
        }

        // Check if student already exists
        const existing = await prisma.student.findUnique({
          where: { studentId: student.studentId }
        })

        if (existing) {
          results.push({
            ...student,
            status: 'error',
            error: 'รหัสนักศึกษาซ้ำ',
          })
          failedCount++
          continue
        }

        // Hash password (default is studentId)
        const hashedPassword = await bcrypt.hash(student.studentId, 10)

        // Create student
        await prisma.student.create({
          data: {
            studentId: student.studentId,
            password: hashedPassword,
            firstName: student.firstName,
            lastName: student.lastName,
            email: student.email || null,
            phone: student.phone || null,
            department: student.department,
            year: student.year,
            isActive: true,
          }
        })

        results.push({
          ...student,
          status: 'success',
        })
        successCount++
      } catch (error: any) {
        results.push({
          ...student,
          status: 'error',
          error: error.message || 'เกิดข้อผิดพลาด',
        })
        failedCount++
      }
    }

    return NextResponse.json({
      success: successCount,
      failed: failedCount,
      total: students.length,
      results,
    })
  } catch (error) {
    console.error('Import students error:', error)
    return NextResponse.json({ error: 'Internal error' }, { status: 500 })
  }
}
```
