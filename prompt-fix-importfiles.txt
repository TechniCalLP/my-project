ข้อสำคัญ: ไฟล์นี้มีรหัส 69xxx ที่เข้าปี 2569 แต่ปัจจุบัน 2567 → ยังไม่เข้า
สูตร:
studentId prefix = 69 → enrolled year 2569
current year = 2567
year = 2567 - 2569 + 1 = -1 → limit to 1
Function:
typescriptfunction calculateYear(studentId: string): number {
  const currentYear = new Date().getFullYear() + 543 // Thai year (2567)
  const yearPrefix = parseInt(studentId.substring(0, 2))
  const enrollYear = 2500 + yearPrefix // 69 → 2569
  const year = currentYear - enrollYear + 1 // 2567 - 2569 + 1 = -1
  return Math.max(1, Math.min(4, year)) // Limit 1-4, so -1 → 1
}

typescriptfunction calculateYear(studentId: string): number {
  const currentYear = new Date().getFullYear() + 543 // 2567
  const yearPrefix = parseInt(studentId.substring(0, 2))
  const enrollYear = 2500 + yearPrefix
  const year = currentYear - enrollYear + 1
  return Math.max(1, Math.min(4, year))
}
C. Function detectDelimiter
typescriptfunction detectDelimiter(text: string): string {
  const firstLine = text.split('\n')[0]
  const commaCount = (firstLine.match(/,/g) || []).length
  const tabCount = (firstLine.match(/\t/g) || []).length
  return tabCount > commaCount ? '\t' : ','
}
D. Function decodeThaiText (decode windows-874)
typescriptfunction decodeThaiText(text: string): string {
  try {
    // ถ้าเป็น buffer ให้ decode โดยใช้ iconv-lite
    if (typeof text === 'string') {
      return text
    }
    return iconv.decode(Buffer.from(text), 'win874')
  } catch {
    return text
  }
}
E. Function parseCSV
typescriptfunction parseCSV(text: string): { students: StudentRow[]; errors: any[] } {
  // Decode windows-874 to UTF-8
  let decodedText = text
  try {
    decodedText = iconv.decode(iconv.encode(text, 'binary'), 'win874')
  } catch (e) {
    console.log('Encoding fallback: using original text')
  }

  // Remove BOM
  decodedText = decodedText.replace(/^\uFEFF/, '')
  
  // Detect delimiter
  const delimiter = detectDelimiter(decodedText)
  const lines = decodedText.split('\n').filter(line => line.trim())
  
  if (lines.length < 2) {
    throw new Error('ไฟล์ว่างเปล่า')
  }

  const students: StudentRow[] = []
  const errors: any[] = []

  // Skip header row
  for (let i = 1; i < lines.length; i++) {
    const line = lines[i].trim()
    if (!line) continue

    const values = line.split(delimiter).map(v => v.trim())

    try {
      if (values.length < 4) {
        throw new Error('คอลัมน์ไม่ครบ')
      }

      // Map columns
      // A (0) = เลขที่ → SKIP
      // B (1) = รหัสนักศึกษา
      // C (2) = ชื่อ
      // D (3) = นามสกุล
      // E (4) = แผนก (อาจว่าง)
      // F-Y (5-24) = SKIP

      const studentId = values[1]?.trim()
      const firstName = values[2]?.trim() || '-'
      const lastName = values[3]?.trim() || '-'
      const department = values[4]?.trim() || 'ทั่วไป'

      // Validate
      if (!studentId || studentId.length < 8) {
        throw new Error('รหัสนักศึกษาไม่ถูกต้อง')
      }

      if (!/^\d{11}$/.test(studentId)) {
        throw new Error('รหัสต้อง 11 หลัก')
      }

      // Calculate year
      const year = calculateYear(studentId)

      students.push({
        studentId,
        firstName,
        lastName,
        department,
        year
      })

    } catch (error: any) {
      errors.push({
        row: i + 1,
        error: error.message,
        data: values
      })
    }
  }

  return { students, errors }
}
F. POST Handler ให้รองรับ Preview + Import
typescriptexport async function POST(req: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    
    if (!session || session.user.role !== 'admin') {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const formData = await req.formData()
    const file = formData.get('file') as File
    const mode = formData.get('mode') as string
    
    if (!file) {
      return NextResponse.json({ error: 'ไม่พบไฟล์' }, { status: 400 })
    }

    // Read file
    const text = await file.text()
    
    // Parse
    const { students, errors } = parseCSV(text)

    // PREVIEW mode
    if (mode === 'preview') {
      return NextResponse.json({
        preview: students.slice(0, 10),
        total: students.length,
        errors: errors.slice(0, 5)
      })
    }

    // IMPORT mode
    const results = {
      success: [],
      errors: [...errors],
      summary: {
        total: students.length,
        success: 0,
        failed: errors.length
      }
    }

    for (const student of students) {
      try {
        // Check duplicate
        const existing = await prisma.student.findUnique({
          where: { studentId: student.studentId }
        })

        if (existing) {
          results.errors.push({
            row: 0,
            error: `รหัส ${student.studentId} มีอยู่แล้ว`,
            data: student
          })
          results.summary.failed++
          continue
        }

        // Hash password
        const hashedPassword = await bcrypt.hash(student.studentId, 10)

        // Insert
        await prisma.student.create({
          data: {
            studentId: student.studentId,
            password: hashedPassword,
            firstName: student.firstName,
            lastName: student.lastName,
            department: student.department,
            year: student.year,
            isActive: true
          }
        })

        results.success.push(student)
        results.summary.success++

      } catch (error: any) {
        results.errors.push({
          row: 0,
          error: error.message,
          data: student
        })
        results.summary.failed++
      }
    }

    return NextResponse.json(results)

  } catch (error: any) {
    console.error('Import error:', error)
    return NextResponse.json({ error: error.message }, { status: 500 })
  }
}
