Step 2: Add Copy Function
typescriptconst handleCopyCode = async (code: string) => {
  try {
    // Copy to clipboard
    await navigator.clipboard.writeText(code)
    
    // Show success toast
    toast({
      title: "Copied!",
      description: `Code ${code} copied to clipboard`,
      duration: 2000,
    })
  } catch (error) {
    console.error('Failed to copy:', error)
    toast({
      title: "Error",
      description: "Failed to copy code",
      variant: "destructive",
    })
  }
}
Step 3: Add Copy Button in Table/List
typescript// In the codes table/list rendering

    {/* Copy Button */}
    <button
