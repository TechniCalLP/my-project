Step 2: Add Download Handler
typescriptconst handleDownload = async (url: string, filename: string) => {
  try {
    const response = await fetch(url)
    const blob = await response.blob()
    
    // Create download link
    const downloadUrl = window.URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = downloadUrl
    link.download = filename
    document.body.appendChild(link)
    
    // Trigger download
    link.click()
    
    // Cleanup
    document.body.removeChild(link)
    window.URL.revokeObjectURL(downloadUrl)
    
    // Show success toast
    toast({
      title: "Success!",
      description: `${filename} downloaded successfully`,
      duration: 3000,
    })
  } catch (error) {
    console.error('Download failed:', error)
    toast({
      title: "Error",
      description: "Failed to download file",
      variant: "destructive",
      duration: 3000,
    })
  }
}

typescriptconst handlePrintDownload = async (activityId: string) => {

Step 5: Alternative - Browser Download API (Simpler)
If the export endpoint returns file directly:
typescriptconst handleExportClick = async (url: string, filename: string) => {
  try {
    // Fetch file
    const response = await fetch(url)
    if (!response.ok) throw new Error('Download failed')
    
    const blob = await response.blob()
    
    // Create temporary link and click it
    const link = document.createElement('a')
    link.href = URL.createObjectURL(blob)
    link.download = filename
    document.body.appendChild(link)
    link.click()
    
    // Cleanup
    setTimeout(() => {
      document.body.removeChild(link)
      URL.revokeObjectURL(link.href)
    }, 100)
    
    // Show notification
    toast({
      title: "✓ Download Complete",
      description: `${filename} saved to Downloads`,
      duration: 2500,
    })
  } catch (error) {
    console.error('Download error:', error)
    toast({
      title: "✗ Download Failed",
      description: "Please try again",
      variant: "destructive",
      duration: 3000,
    })
  }
}
