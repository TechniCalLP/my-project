```typescript
// components/dashboard/activity-card.tsx
export function ActivityCard({ activity }) {
  const isOpen = activity.status === 'ACTIVE'
  const dateRange = formatThaiDateRange(activity.startDate, activity.endDate)
  
  return (
    
      
        
        
          
            {activity.title}

            >

          )}

}

function formatThaiDateRange(startDate: Date, endDate: Date) {
  const thaiMonths = [
    'มกราคม', 'กุมภาพันธ์', 'มีนาคม', 'เมษายน', 'พฤษภาคม', 'มิถุนายน',
    'กรกฎาคม', 'สิงหาคม', 'กันยายน', 'ตุลาคม', 'พฤศจิกายน', 'ธันวาคม'
  ]
  
  const start = new Date(startDate)
  const end = new Date(endDate)
  
  const startDay = start.getDate()
  const startMonth = thaiMonths[start.getMonth()]
  const startYear = start.getFullYear() + 543
  
  const endDay = end.getDate()
  const endMonth = thaiMonths[end.getMonth()]
  const endYear = end.getFullYear() + 543
  
  if (start.toDateString() === end.toDateString()) {
    return `${startDay} ${startMonth} ${startYear}`
  }
  
  if (start.getMonth() === end.getMonth() && start.getFullYear() === end.getFullYear()) {
    return `${startDay}-${endDay} ${startMonth} ${startYear}`
  }
  
  return `${startDay} ${startMonth} ${startYear} - ${endDay} ${endMonth} ${endYear}`
}
```

## 3. Database Schema Check

Ensure Activity model has date fields:

```prisma
model Activity {
  id                String   @id @default(cuid())
  title             String
  description       String?
  year              String
  semester          String
  category          Category
  targetDepartments String[]
  status            ActivityStatus @default(ACTIVE)
  startDate         DateTime  // Required
  endDate           DateTime  // Required
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt
  
  codes            ActivityCode[]
  participations   Participation[]
  admin            Admin?    @relation(fields: [adminId], references: [id])
  adminId          String?
}

enum ActivityStatus {
  ACTIVE      // เปิดรับสมัคร
  COMPLETED   // ปิดแล้ว
  CANCELLED   // ยกเลิก
}
```
