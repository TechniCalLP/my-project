## 1. Update Code Generation Logic

File: `app/api/admin/activities/route.ts`

Replace the code generation section with:

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/app/api/auth/[...nextauth]/route'
import { prisma } from '@/lib/prisma'

// Category prefix mapping
const CATEGORY_PREFIX: Record = {
  ACADEMIC: 'ACT',
  SPORTS: 'SPT',
  ARTS: 'ART',
  VOLUNTEER: 'VOL',
  MORALITY: 'MOR',
  HEALTH: 'HLT',
}

export async function POST(req: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    if (!session || session.user.role !== 'admin') {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const data = await req.json()

    // Create activity
    const activity = await prisma.activity.create({
      data: {
        title: data.title,
        description: data.description,
        category: data.category,
        year: data.year,
        semester: data.semester,
        startDate: new Date(data.startDate),
        endDate: new Date(data.endDate),
        status: data.status,
      }
    })

    // Generate activity codes with mixed language format
    const codeCount = data.codeCount || 5
    const prefix = CATEGORY_PREFIX[data.category] || 'ACT'
    const codes = []
    
    for (let i = 0; i < codeCount; i++) {
      // Generate unique code: PREFIX + 3-digit number (e.g., ACT001, SPT045)
      const number = String(i + 1).padStart(3, '0')
      const code = `${prefix}${number}`
      
      codes.push({
        code,
        activityId: activity.id,
        isUsed: false,
      })
    }

    await prisma.activityCode.createMany({
      data: codes
    })

    return NextResponse.json(activity)
  } catch (error) {
    console.error('Create activity error:', error)
    return NextResponse.json({ error: 'Internal error' }, { status: 500 })
  }
}
```

```typescript
}
```

```typescript
  const router = useRouter()
  const { toast } = useToast()
  const [loading, setLoading] = useState(false)

  const handleDelete = async () => {
    if (!confirm(`ต้องการลบรหัส ${code} ใช่หรือไม่?`)) {
      return
    }

    setLoading(true)

    try {
      const response = await fetch(`/api/admin/codes/${codeId}`, {
        method: 'DELETE',

        title: "ลบสำเร็จ",

        description: "ไม่สามารถลบรหัสได้",

    } finally {
      setLoading(false)
    }
  }
```

## 6. Delete Code API

File: `app/api/admin/codes/[id]/route.ts`

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/app/api/auth/[...nextauth]/route'
import { prisma } from '@/lib/prisma'

export async function DELETE(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session || session.user.role !== 'admin') {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    // Check if code is already used
    const code = await prisma.activityCode.findUnique({
      where: { id: params.id }
    })

    if (!code) {
      return NextResponse.json({ error: 'Code not found' }, { status: 404 })
    }

    if (code.isUsed) {
      return NextResponse.json({ error: 'Cannot delete used code' }, { status: 400 })
    }

    await prisma.activityCode.delete({
      where: { id: params.id }
    })

    return NextResponse.json({ success: true })
  } catch (error) {
    console.error('Delete code error:', error)
    return NextResponse.json({ error: 'Internal error' }, { status: 500 })
  }
}
```
