### Data Fetching
```typescript
export default async function Dashboard() {
  const session = await getServerSession(authOptions)
  const student = session.user
  
  // Stats
  const joined = await prisma.participation.count({
    where: { studentId: student.studentId }
  })
  
  const total = await prisma.activity.count({
    where: {
      year: student.year,
      status: 'ACTIVE',
      OR: [
        { targetDepartments: { isEmpty: true } },
        { targetDepartments: { has: student.department } }
      ]
    }
  })
  
  const remaining = total - joined
  const progress = Math.round((joined / total) * 100)
  
  // Upcoming activities (not joined yet)
  const upcoming = await prisma.activity.findMany({
    where: {
      year: student.year,
      status: 'ACTIVE',
      OR: [
        { targetDepartments: { isEmpty: true } },
        { targetDepartments: { has: student.department } }
      ],
      NOT: {
        participations: {
          some: { studentId: student.studentId }
        }
      }
    },
    take: 5,
    orderBy: { createdAt: 'desc' }
  })
  
  return <DashboardView stats={{ joined, remaining, progress }} upcoming={upcoming} />
}
```

```typescript
export default async function ActivitiesPage({ searchParams }) {
  const session = await getServerSession(authOptions)
  const student = session.user
  
  const semester = searchParams.semester || 'all'
  const category = searchParams.category || 'all'
  const status = searchParams.status || 'ACTIVE'
  
  const activities = await prisma.activity.findMany({
    where: {
      year: student.year, // Always filter by student year
      ...(semester !== 'all' && { semester }),
      ...(category !== 'all' && { category }),
      status,
      OR: [
        { targetDepartments: { isEmpty: true } },
        { targetDepartments: { has: student.department } }
      ]
    },
    include: {
      participations: {
        where: { studentId: student.studentId }
      }
    }
  })
  
  return <ActivitiesGrid activities={activities} />
}
```

## 8. API Routes

### GET /api/student/stats
```typescript
export async function GET(req: Request) {
  const session = await getServerSession(authOptions)
  
  const joined = await prisma.participation.count({
    where: { studentId: session.user.studentId }
  })
  
  const total = await prisma.activity.count({
    where: { year: session.user.year, status: 'ACTIVE' }
  })
  
  return Response.json({ joined, remaining: total - joined, progress: (joined/total)*100 })
}
```

### POST /api/activities/join
```typescript
export async function POST(req: Request) {
  const { code } = await req.json()
  const session = await getServerSession(authOptions)
  
  // Find code
  const activityCode = await prisma.activityCode.findUnique({
    where: { code },
    include: { activity: true }
  })
  
  if (!activityCode) {
    return Response.json({ error: 'รหัสไม่ถูกต้อง' }, { status: 404 })
  }
  
  // Check already joined
  const existing = await prisma.participation.findUnique({
    where: {
      studentId_activityId: {
        studentId: session.user.studentId,
        activityId: activityCode.activityId
      }
    }
  })
  
  if (existing) {
    return Response.json({ error: 'คุณเข้าร่วมกิจกรรมนี้แล้ว' }, { status: 400 })
  }
  
  // Create participation
  const participation = await prisma.participation.create({
    data: {
      studentId: session.user.studentId,
      activityId: activityCode.activityId,
      codeUsed: code
    }
  })
  
  return Response.json(participation)
}
```

### GET /api/activities
```typescript
export async function GET(req: Request) {
  const session = await getServerSession(authOptions)
  const { searchParams } = new URL(req.url)
  
  const semester = searchParams.get('semester')
  const category = searchParams.get('category')
  
  const activities = await prisma.activity.findMany({
    where: {
      year: session.user.year, // Auto-filter
      ...(semester && { semester }),
      ...(category && { category }),
      status: 'ACTIVE',
      OR: [
        { targetDepartments: { isEmpty: true } },
        { targetDepartments: { has: session.user.department } }
      ]
    },
    include: {
      participations: {
        where: { studentId: session.user.studentId }
      }
    }
  })
  
  return Response.json(activities)
}
```
