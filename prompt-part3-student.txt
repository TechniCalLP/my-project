# Student Activity System - Part 2: Modern Student Experience

## Critical Design Decisions

### Layout Philosophy
NO SIDEBAR - Use modern top navigation pattern

Desktop: Horizontal nav bar
Mobile: Bottom tab bar
Content: Full width (no wasted space)
Navigation: 4 items only (หน้าหลัก, กิจกรรม, ประวัติ, ตั้งค่า)


### What NOT to Build
❌ Sidebar navigation (wastes horizontal space)
❌ Quick action cards (redundant with nav)
❌ Admin features
❌ "All activities" view (only student's year)
❌ Generic dashboard (must be personalized)

```typescript
// app/(student)/layout.tsx
import { getServerSession } from "next-auth"
import { redirect } from "next/navigation"
import { TopNav } from "@/components/layout/top-nav"
import { BottomNav } from "@/components/layout/bottom-nav"
import { MobileHeader } from "@/components/layout/mobile-header"

export default async function StudentLayout({
  children,
}: {
  children: React.ReactNode
}) {
  const session = await getServerSession()
  
  if (!session) {
    redirect("/login")
  }
  
  // Redirect to settings if first login
  if (session.user.isFirstLogin && pathname !== "/settings") {
    redirect("/settings")
  }

  return (
    <div className="min-h-screen bg-gray-50">
      {/* Desktop Top Navigation */}
      <TopNav className="hidden md:flex" user={session.user} />
      
      {/* Mobile Header */}
      <MobileHeader className="md:hidden" user={session.user} />
      
      {/* Main Content - Full Width */}
      <main className="pb-20 md:pb-0">
        <div className="max-w-7xl mx-auto px-4 md:px-6 py-6 md:py-8">
          {children}
        </div>
      </main>
      
      {/* Mobile Bottom Navigation */}
      <BottomNav className="md:hidden" />
    </div>
  )
}
```

## 2. Login Page

```typescript
// app/(auth)/login/page.tsx
"use client"

import { useState } from "react"
import { signIn } from "next-auth/react"
import { useRouter } from "next/navigation"
import { School, Eye, EyeOff } from "lucide-react"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Label } from "@/components/ui/label"
import { Card, CardContent, CardHeader } from "@/components/ui/card"
import { useToast } from "@/components/ui/use-toast"

export default function LoginPage() {
  const router = useRouter()
  const { toast } = useToast()
  const [loading, setLoading] = useState(false)
  const [showPassword, setShowPassword] = useState(false)

  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault()
    setLoading(true)

    const formData = new FormData(e.currentTarget)
    const studentId = formData.get("studentId") as string
    const password = formData.get("password") as string

    // Validate student ID (11 digits)
    if (!/^\d{11}$/.test(studentId)) {
      toast({
        title: "ข้อมูลไม่ถูกต้อง",
        description: "รหัสนักศึกษาต้องเป็นตัวเลข 11 หลัก",
        variant: "destructive",
      })
      setLoading(false)
      return
    }

    try {
      const result = await signIn("credentials", {
        studentId,
        password,
        redirect: false,
      })

      if (result?.ok) {
        router.push("/dashboard")
        router.refresh()
      } else {
        toast({
          title: "เข้าสู่ระบบไม่สำเร็จ",
          description: "รหัสนักศึกษาหรือรหัสผ่านไม่ถูกต้อง",
          variant: "destructive",
        })
      }
    } catch (error) {
      toast({
        title: "เกิดข้อผิดพลาด",
        description: "กรุณาลองใหม่อีกครั้ง",
        variant: "destructive",
      })
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="min-h-screen bg-gradient-to-br from-blue-50 via-white to-indigo-50 flex items-center justify-center p-4">
      <Card className="w-full max-w-md shadow-xl">
        <CardHeader className="text-center space-y-4 pt-8 pb-6">
          {/* Logo */}
          <div className="mx-auto w-20 h-20 bg-blue-100 rounded-full flex items-center justify-center">
            <School className="w-10 h-10 text-blue-600" />
          </div>
          
          {/* Title */}
          <div>
            <h1 className="text-2xl font-bold text-gray-900">
              วิทยาลัยเทคนิคลำปาง
            </h1>
            <p className="text-gray-600 mt-1">
              ระบบบันทึกกิจกรรมนักศึกษา
            </p>
          </div>
        </CardHeader>

        <CardContent className="pb-8">
          <form onSubmit={handleSubmit} className="space-y-6">
            {/* Student ID Input */}
            <div className="space-y-2">
              <Label htmlFor="studentId">รหัสนักศึกษา</Label>
              <Input
                id="studentId"
                name="studentId"
                type="text"
                placeholder="11 หลัก"
                maxLength={11}
                required
                className="h-12 text-base"
                autoComplete="username"
              />
            </div>

            {/* Password Input */}
            <div className="space-y-2">
              <Label htmlFor="password">รหัสผ่าน</Label>
              <div className="relative">
                <Input
                  id="password"
                  name="password"
                  type={showPassword ? "text" : "password"}
                  placeholder="••••••••"
                  required
                  className="h-12 text-base pr-12"
                  autoComplete="current-password"
                />
                <button
                  type="button"
                  onClick={() => setShowPassword(!showPassword)}
                  className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-500 hover:text-gray-700"
                >
                  {showPassword ? (
                    <EyeOff className="w-5 h-5" />
                  ) : (
                    <Eye className="w-5 h-5" />
                  )}
                </button>
              </div>
            </div>

            {/* Submit Button */}
            <Button
              type="submit"
              className="w-full h-12 text-base"
              disabled={loading}
            >
              {loading ? "กำลังเข้าสู่ระบบ..." : "เข้าสู่ระบบ"}
            </Button>

            {/* Help Text */}
            <p className="text-sm text-center text-gray-600">
              ลืมรหัสผ่าน? ติดต่อครูที่ปรึกษา
            </p>
          </form>
        </CardContent>
      </Card>
    </div>
  )
}
```

## 3. Dashboard - Personalized & Actionable

```typescript
// app/(student)/dashboard/page.tsx
import { getServerSession } from "next-auth"
import { prisma } from "@/lib/prisma"
import { ProgressHero } from "@/components/dashboard/progress-hero"
import { StatsCards } from "@/components/dashboard/stats-cards"
import { ActivityList } from "@/components/dashboard/activity-list"
import { RecentHistory } from "@/components/dashboard/recent-history"
import { PageHeader } from "@/components/shared/page-header"
import { Section } from "@/components/shared/section"

export default async function DashboardPage() {
  const session = await getServerSession()
  const student = session.user

  // Get stats
  const joined = await prisma.participation.count({
    where: { studentId: student.studentId }
  })

  const total = await prisma.activity.count({
    where: {
      year: student.year,
      status: 'ACTIVE',
      OR: [
        { targetDepartments: { isEmpty: true } },
        { targetDepartments: { has: student.department } }
      ]
    }
  })

  const remaining = total - joined
  const progress = total > 0 ? Math.round((joined / total) * 100) : 0

  // Get upcoming activities (not joined yet)
  const upcomingActivities = await prisma.activity.findMany({
    where: {
      year: student.year,
      status: 'ACTIVE',
      OR: [
        { targetDepartments: { isEmpty: true } },
        { targetDepartments: { has: student.department } }
      ],
      NOT: {
        participations: {
          some: { studentId: student.studentId }
        }
      }
    },
    take: 5,
    orderBy: { createdAt: 'desc' }
  })

  // Get recent history
  const recentHistory = await prisma.participation.findMany({
    where: { studentId: student.studentId },
    include: { activity: true },
    orderBy: { joinedAt: 'desc' },
    take: 3
  })

  return (
    <div className="space-y-8">
      {/* Welcome Header */}
      <PageHeader
        title={`สวัสดี, ${student.firstName} ${student.lastName} 👋`}
        subtitle={`${student.year} • ${student.department} • เทอม ${getCurrentSemester()}/2567`}
      />

      {/* Progress Hero */}
      <ProgressHero
        joined={joined}
        total={total}
        remaining={remaining}
        progress={progress}
      />

      {/* Stats Cards */}
      <StatsCards
        joined={joined}
        remaining={remaining}
        progress={progress}
      />

      {/* Upcoming Activities */}
      {upcomingActivities.length > 0 && (
        <Section
          title="กิจกรรมที่ยังไม่ได้เข้าร่วม"
          description={`${upcomingActivities.length} กิจกรรม`}
          action={{
            label: "ดูทั้งหมด",
            href: "/activities"
          }}
        >
          <ActivityList activities={upcomingActivities} variant="upcoming" />
        </Section>
      )}

      {/* Recent History */}
      {recentHistory.length > 0 && (
        <Section
          title="กิจกรรมที่เข้าร่วมล่าสุด"
          action={{
            label: "ดูประวัติ",
            href: "/history"
          }}
        >
          <RecentHistory items={recentHistory} />
        </Section>
      )}

      {/* Empty State if no activities */}
      {upcomingActivities.length === 0 && recentHistory.length === 0 && (
        <EmptyState
          icon={Calendar}
          title="ยังไม่มีกิจกรรม"
          description="ขณะนี้ยังไม่มีกิจกรรมสำหรับคุณ"
        />
      )}
    </div>
  )
}

function getCurrentSemester() {
  const month = new Date().getMonth() + 1
  return month >= 6 && month <= 10 ? 1 : 2
}
```
