# Login Page - Split Screen Design

## Design Concept

Create a modern split-screen login page with brand colors:
- Left side: Navy (#2E3192) to Red (#C52833) gradient with branding
- Right side: White background with login form

## Layout
┌──────────────────┬──────────────────┐
│                  │                  │

│  (Gradient)      │  (White)         │
│                  │                  │
│  ┌──────────┐   │  ┌────────────┐ │
│  │  Logo    │   │  │            │ │

## Implementation

File: `app/(auth)/login/page.tsx`

```typescript
"use client"

import { useState } from "react"
import { signIn } from "next-auth/react"
import { useRouter } from "next/navigation"
import Image from "next/image"
import { Eye, EyeOff, GraduationCap } from "lucide-react"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Label } from "@/components/ui/label"
import { useToast } from "@/components/ui/use-toast"

export default function LoginPage() {
  const router = useRouter()
  const { toast } = useToast()
  const [loading, setLoading] = useState(false)
  const [showPassword, setShowPassword] = useState(false)

  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault()
    setLoading(true)

    const formData = new FormData(e.currentTarget)
    const studentId = formData.get("studentId") as string
    const password = formData.get("password") as string

    if (!/^\d{11}$/.test(studentId)) {
      toast({
        title: "ข้อมูลไม่ถูกต้อง",
        description: "รหัสนักศึกษาต้องเป็นตัวเลข 11 หลัก",
        variant: "destructive",
      })
      setLoading(false)
      return
    }

    try {
      const result = await signIn("credentials", {
        studentId,
        password,
        redirect: false,
      })

      if (result?.ok) {
        router.push("/dashboard")
        router.refresh()
      } else {
        toast({
          title: "เข้าสู่ระบบไม่สำเร็จ",
          description: "รหัสนักศึกษาหรือรหัสผ่านไม่ถูกต้อง",
          variant: "destructive",
        })
      }
    } catch (error) {
      toast({
        title: "เกิดข้อผิดพลาด",
        description: "กรุณาลองใหม่อีกครั้ง",
        variant: "destructive",
      })
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="min-h-screen flex">
      {/* LEFT SIDE - Brand Gradient */}
      <div className="hidden lg:flex lg:w-1/2 bg-gradient-to-br from-primary via-primary-600 to-secondary relative overflow-hidden">
        {/* Decorative background elements */}
        <div className="absolute inset-0">
          <div className="absolute top-20 left-20 w-64 h-64 bg-white/5 rounded-full blur-3xl" />
          <div className="absolute bottom-20 right-20 w-80 h-80 bg-white/10 rounded-full blur-3xl" />
          <div className="absolute top-1/2 left-1/4 w-48 h-48 bg-secondary/20 rounded-full blur-2xl" />
        </div>

        {/* Content */}
        <div className="relative z-10 flex flex-col items-center justify-center w-full p-12 text-white">
          {/* Logo */}
          <div className="mb-8">
            <div className="w-40 h-40 bg-white/10 backdrop-blur-sm rounded-full flex items-center justify-center border-4 border-white/20 shadow-2xl">
              {/* Replace with actual logo */}
              <Image 
                src="/logo-college.png" 
                alt="วิทยาลัยเทคนิคลำปาง"
                width={120}
                height={120}
                className="rounded-full"
                priority
              />
            </div>
          </div>

          {/* Title */}
          <h1 className="text-4xl font-bold text-center mb-4">
            วิทยาลัยเทคนิคลำปาง
          </h1>
          
          <p className="text-xl text-white/90 text-center mb-8">
            Lampang Technical College
          </p>

          {/* Tagline */}
          <div className="bg-white/10 backdrop-blur-sm px-8 py-4 rounded-full border border-white/20">
            <p className="text-lg font-medium">
              ระบบบันทึกกิจกรรมนักศึกษา
            </p>
          </div>

          {/* Additional info */}
          <div className="mt-12 text-center">
            <p className="text-white/80 text-sm">
              Student Activity Management System
            </p>
            <p className="text-white/60 text-xs mt-2">
              Academic Year 2567
            </p>
          </div>
        </div>
      </div>

      {/* RIGHT SIDE - Login Form */}
      <div className="w-full lg:w-1/2 flex items-center justify-center p-8 bg-gray-50">
        <div className="w-full max-w-md">
          {/* Mobile Logo (shown only on small screens) */}
          <div className="lg:hidden text-center mb-8">
            <div className="w-20 h-20 bg-gradient-to-br from-primary to-secondary rounded-full flex items-center justify-center mx-auto mb-4 shadow-lg">
              <Image 
                src="/logo-college.png" 
                alt="วิทยาลัยเทคนิคลำปาง"
                width={64}
                height={64}
                className="rounded-full"
                priority
              />
            </div>
            <h2 className="text-xl font-bold text-gray-900">
              วิทยาลัยเทคนิคลำปาง
            </h2>
            <p className="text-gray-600 text-sm mt-1">
              ระบบบันทึกกิจกรรมนักศึกษา
            </p>
          </div>

          {/* Login Card */}
          <div className="bg-white rounded-2xl shadow-xl p-8 lg:p-10">
            {/* Header */}
            <div className="mb-8">
              <h2 className="text-2xl font-bold text-gray-900 mb-2">
                เข้าสู่ระบบ
              </h2>
              <p className="text-gray-600">
                กรุณากรอกข้อมูลเพื่อเข้าใช้งาน
              </p>
            </div>

            {/* Form */}
            <form onSubmit={handleSubmit} className="space-y-6">
              {/* Student ID */}
              <div className="space-y-2">
                <Label 
                  htmlFor="studentId" 
                  className="text-gray-700 font-medium text-sm"
                >
                  รหัสนักศึกษา
                </Label>
                <Input
                  id="studentId"
                  name="studentId"
                  type="text"
                  placeholder="กรอกรหัสนักศึกษา 11 หลัก"
                  maxLength={11}
                  required
                  className="h-12 text-base border-gray-300 focus:border-primary focus:ring-primary"
                  autoComplete="username"
                />
              </div>

              {/* Password */}
              <div className="space-y-2">
                <Label 
                  htmlFor="password" 
                  className="text-gray-700 font-medium text-sm"
                >
                  รหัสผ่าน
                </Label>
                <div className="relative">
                  <Input
                    id="password"
                    name="password"
                    type={showPassword ? "text" : "password"}
                    placeholder="กรอกรหัสผ่าน"
                    required
                    className="h-12 text-base pr-12 border-gray-300 focus:border-primary focus:ring-primary"
                    autoComplete="current-password"
                  />
                  <button
                    type="button"
                    onClick={() => setShowPassword(!showPassword)}
                    className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-500 hover:text-primary transition-colors"
                  >
                    {showPassword ? (
                      <EyeOff className="w-5 h-5" />
                    ) : (
                      <Eye className="w-5 h-5" />
                    )}
                  </button>
                </div>
              </div>

              {/* Submit Button - NAVY */}
              <Button
                type="submit"
                className="w-full h-12 text-base font-semibold bg-primary hover:bg-primary-700 text-white shadow-lg hover:shadow-xl transition-all duration-200"
                disabled={loading}
              >
                {loading ? (
                  <span className="flex items-center gap-2">
                    <svg className="animate-spin h-5 w-5" viewBox="0 0 24 24">
                      <circle 
                        className="opacity-25" 
                        cx="12" 
                        cy="12" 
                        r="10" 
                        stroke="currentColor" 
                        strokeWidth="4" 
                        fill="none" 
                      />
                      <path 
                        className="opacity-75" 
                        fill="currentColor" 
                        d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z" 
                      />
                    </svg>
                    กำลังเข้าสู่ระบบ...
                  </span>
                ) : (
                  "เข้าสู่ระบบ"
                )}
              </Button>

              {/* Help Link */}
              <div className="text-center pt-4">
                <p className="text-sm text-gray-600">
                  ลืมรหัสผ่าน?{" "}
                  <button 
                    type="button"
                    className="text-primary hover:text-primary-700 font-semibold hover:underline"
                  >
                    ติดต่อครูที่ปรึกษา
                  </button>
                </p>
              </div>
            </form>
          </div>

          {/* Footer */}
          <div className="mt-8 text-center">
            <p className="text-xs text-gray-500">
              © 2024 วิทยาลัยเทคนิคลำปาง. All rights reserved.
            </p>
          </div>
        </div>
      </div>
    </div>
  )
}
```

## Key Features

### Left Side (Gradient Panel)
1. **Background**: Navy → Red gradient (from-primary to-secondary)
2. **Decorative Elements**: Floating gradient circles with blur
3. **Logo**: Large 160px circle with white border
4. **College Name**: Large white text (4xl)
5. **Tagline**: Rounded pill with backdrop blur
6. **Additional Info**: Academic year, system name

### Right Side (Form Panel)
1. **Background**: Light gray (bg-gray-50)
2. **Card**: White with shadow and rounded corners
3. **Form**: Clean inputs with navy focus states
4. **Button**: Navy color with hover effect
5. **Mobile**: Shows small logo on small screens

### Responsive Design
```css
Desktop (≥1024px):
- Left: 50% width (gradient)
- Right: 50% width (form)

Mobile (<1024px):
- Left: Hidden
- Right: Full width
- Shows mobile logo at top
```

### Brand Colors Used
```css
- Background gradient: primary → secondary
- Decorative circles: white/5, white/10, secondary/20
- Logo border: white/20
- Button: bg-primary hover:bg-primary-700
- Focus rings: border-primary ring-primary
- Links: text-primary hover:text-primary-700
```

### Visual Elements
1. Floating gradient orbs (3 circles)
2. Backdrop blur on logo container
3. Border on tagline pill
4. Large shadows on card
5. Smooth transitions on all interactions
