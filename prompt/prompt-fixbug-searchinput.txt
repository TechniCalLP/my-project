Implementation
Step 1: Add State for Search
typescriptconst [searchTerm, setSearchTerm] = useState('')
const [filteredActivities, setFilteredActivities] = useState([])

// Filter activities when search term changes
useEffect(() => {
  if (!activities) return
  
  const filtered = activities.filter((activity) =>
    activity.name.toLowerCase().includes(searchTerm.toLowerCase())
  )
  
  setFilteredActivities(filtered)
}, [searchTerm, activities])
Step 2: Add Search Input UI
typescript<div className="mb-4">
  <input
    type="text"
    placeholder="ค้นหากิจกรรม..."
    value={searchTerm}
    onChange={(e) => setSearchTerm(e.target.value)}
    className="w-full px-4 py-2 border rounded-lg"
  />
</div>
Step 3: Display Filtered Results
typescript// Use filteredActivities instead of activities

      ไม่พบกิจกรรม
    </td>
  </tr>
)}
Step 4: Show Result Count
